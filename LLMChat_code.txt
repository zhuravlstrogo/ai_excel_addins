Private Sub FillModels()
    Dim m As Variant
    cboModel.Clear
    For Each m In Split(Setting("ModelList"), ",")
        If Len(Trim$(m)) > 0 Then cboModel.AddItem Trim$(m)
    Next m
    cboModel.Text = Setting("Model")
End Sub

' Adds the drag border to the window MSForms created. Safe to call more than once;
' if the API call fails the form simply stays fixed-size.
Private Sub MakeResizable()
    #If VBA7 Then
        Dim hWnd As LongPtr, st As LongPtr
    #Else
        Dim hWnd As Long, st As Long
    #End If
    On Error GoTo Done
    hWnd = FindWindowA("ThunderDFrame", Me.Caption)
    If hWnd = 0 Then hWnd = FindWindowA("ThunderXFrame", Me.Caption)
    If hWnd = 0 Then GoTo Done
    st = GetWinLong(hWnd, GWL_STYLE)
    SetWinLong hWnd, GWL_STYLE, st Or WS_THICKFRAME Or WS_MAXIMIZEBOX
    DrawMenuBar hWnd
Done:
    On Error GoTo 0
    LayoutControls
End Sub

Private Sub RestoreSize()
    Dim w As Single, h As Single
    w = Val(Setting("FormWidth"))
    h = Val(Setting("FormHeight"))
    If w >= MIN_W Then Me.Width = w
    If h >= MIN_H Then Me.Height = h
End Sub

Private Sub SaveSize()
    On Error Resume Next
    SaveSettingValue "FormWidth", CStr(CLng(Me.Width))
    SaveSettingValue "FormHeight", CStr(CLng(Me.Height))
End Sub

Private Function Mx(ByVal a As Single, ByVal b As Single) As Single
    If a > b Then Mx = a Else Mx = b
End Function

Private Sub LayoutControls()
    Dim vw As Single, vh As Single, w As Single, h As Single
    Dim colW As Single, col2 As Single, capW As Single
    Dim bw As Single, rowBtn As Single, inTop As Single, inH As Single, chkL As Single

    If txtHistory Is Nothing Then Exit Sub      ' fires once before the controls exist
    If mLayingOut Then Exit Sub
    mLayingOut = True
    On Error GoTo Done

    If Me.Width < MIN_W Then Me.Width = MIN_W
    If Me.Height < MIN_H Then Me.Height = MIN_H

    ' the layout is always built for at least CONTENT_MIN; a smaller window scrolls
    vw = Mx(Me.InsideWidth, CONTENT_MIN_W)
    vh = Mx(Me.InsideHeight, CONTENT_MIN_H)
    Me.KeepScrollBarsVisible = fmScrollBarsNone
    Me.ScrollWidth = CONTENT_MIN_W
    Me.ScrollHeight = CONTENT_MIN_H
    Me.ScrollBars = fmScrollBarsBoth

    w = vw - 12                                 ' usable width, 6 pt margins
    h = vh

    ' narrow window: drop the captions and give the space to the drop-downs
    capW = 42
    If w < 318 Then capW = 0
    lblModel.Visible = (capW > 0)
    lblData.Visible = (capW > 0)
    lblPreset.Visible = (capW > 0)

    ' row 1: model + data, two equal columns
    colW = (w - 6) / 2
    col2 = 6 + colW + 6
    lblModel.Left = 6: lblModel.Top = 9
    cboModel.Left = 6 + capW: cboModel.Top = 6: cboModel.Width = Mx(40, colW - capW)
    lblData.Left = col2: lblData.Top = 9
    cboContext.Left = col2 + capW: cboContext.Top = 6: cboContext.Width = Mx(40, colW - capW)

    ' row 2: preset grows, the checkbox stays pinned to the right
    chkL = 6 + w - 106
    lblPreset.Left = 6: lblPreset.Top = 33
    cboPreset.Left = 6 + capW: cboPreset.Top = 30: cboPreset.Width = Mx(40, chkL - 6 - 6 - capW)
    chkAgent.Left = chkL: chkAgent.Top = 30

    ' the input box gives up height first, the history keeps what is left
    inH = 70
    If vh < 400 Then inH = Mx(28, 70 - (400 - vh))

    ' bottom block: 3 button rows (step 28) + status line, anchored to the bottom
    rowBtn = h - 106
    inTop = rowBtn - inH - 6
    bw = (w - 12) / 3

    txtHistory.Left = 6: txtHistory.Top = 54
    txtHistory.Width = w: txtHistory.Height = Mx(24, inTop - 60)
    txtInput.Left = 6: txtInput.Top = inTop: txtInput.Width = w: txtInput.Height = inH

    PlaceRow btnSend, btnExplain, btnNew, rowBtn, bw
    PlaceRow btnFormula, btnInsert, btnUndo, rowBtn + 28, bw
    PlaceRow btnSaveChat, btnSettings, btnCache, rowBtn + 56, bw
    lblStatus.Left = 6: lblStatus.Top = rowBtn + 86: lblStatus.Width = w

    LayoutSettings w, h
Done:
    mLayingOut = False
End Sub

Private Sub PlaceRow(ByVal b1 As MSForms.CommandButton, ByVal b2 As MSForms.CommandButton, _
                     ByVal b3 As MSForms.CommandButton, ByVal t As Single, ByVal bw As Single)
    b1.Top = t: b1.Left = 6: b1.Width = bw
    b2.Top = t: b2.Left = 6 + bw + 6: b2.Width = bw
    b3.Top = t: b3.Left = 6 + 2 * (bw + 6): b3.Width = bw
End Sub

Private Sub DoSend(ByVal userText As String, ByVal forceData As Boolean, ByVal clearInput As Boolean)
    Dim ctx As String, ans As String, shown As String, actions As Collection, agent As Boolean
    userText = Trim$(userText)
    If Len(userText) = 0 Then Exit Sub
    agent = CBool(chkAgent.Value)

    ctx = GetContext(forceData, agent)
    If Len(mCtxLabel) > 0 Then
        AppendHistory WhoYou(), userText & RU("   [+ \u0434\u0430\u043D\u043D\u044B\u0435: ") & mCtxLabel & "]"
    Else
        AppendHistory WhoYou(), userText
    End If

    SetBusy True, RU("\u0416\u0434\u0443 \u043E\u0442\u0432\u0435\u0442 \u043C\u043E\u0434\u0435\u043B\u0438...")
    ans = ChatSend(userText, ctx, cboModel.Text, agent)
    SetBusy False, RU("\u0413\u043E\u0442\u043E\u0432\u043E") & TokensText()

    If Left$(ans, 6) = "#ERROR" Then
        AppendHistory WhoAI(), ans
        Exit Sub
    End If
    If clearInput Then txtInput.Text = ""

    Set actions = SplitActions(ans, shown)
    LastAnswer = shown
    AppendHistory WhoAI(), shown

    If actions Is Nothing Then Exit Sub
    If actions.Count = 0 Then Exit Sub
    If Not agent Then
        AppendHistory "Excel", RU("\u041C\u043E\u0434\u0435\u043B\u044C \u043F\u0440\u0435\u0434\u043B\u043E\u0436\u0438\u043B\u0430 \u0438\u0437\u043C\u0435\u043D\u0435\u043D\u0438\u044F, \u043D\u043E \u0433\u0430\u043B\u043E\u0447\u043A\u0430 \u00AB\u0420\u0430\u0437\u0440\u0435\u0448\u0438\u0442\u044C \u043F\u0440\u0430\u0432\u043A\u0438\u00BB \u0432\u044B\u043A\u043B\u044E\u0447\u0435\u043D\u0430.")
        Exit Sub
    End If

    ' no confirmation dialog: the list of changes (with [!] warnings) and the result go to the chat
    AppendHistory "Excel", RU("\u041F\u0440\u0435\u0434\u043B\u043E\u0436\u0435\u043D\u043D\u044B\u0435 \u0438\u0437\u043C\u0435\u043D\u0435\u043D\u0438\u044F:") & ActionsPreview(actions)
    AppendHistory "Excel", ApplyActions(actions)
End Sub

Private Function GetContext(ByVal forceData As Boolean, ByVal agent As Boolean) As String
    Dim mode As Long, rng As Range, data As String
    mCtxLabel = ""
    mode = cboContext.ListIndex      ' 0 selection, 1 sheet, 2 workbook overview, 3 none
    If forceData And mode = 3 Then mode = 0

    On Error Resume Next
    Select Case mode
        Case 0
            Set rng = ActiveWindow.RangeSelection
            data = BuildContext(rng, True, True)
            If Len(data) > 0 Then mCtxLabel = rng.Worksheet.Name & "!" & rng.Address(False, False)
        Case 1
            Set rng = ActiveSheet.UsedRange
            data = BuildContext(rng, True, True)
            If Len(data) > 0 Then mCtxLabel = RU("\u043B\u0438\u0441\u0442 ") & rng.Worksheet.Name
        Case 2
            data = BuildWorkbookOverview()
            If Len(data) > 0 Then mCtxLabel = RU("\u043E\u0431\u0437\u043E\u0440 \u043A\u043D\u0438\u0433\u0438")
    End Select
    On Error GoTo 0

    If Len(data) > 0 Or agent Then
        GetContext = WorkbookInfoLine() & vbLf & data
    End If
End Function

Private Sub AppendHistory(ByVal who As String, ByVal text As String)
    Dim s As String, t As String, p As Long
    s = Replace(Replace(text, vbCrLf, vbLf), vbLf, vbCrLf)
    t = txtHistory.Text
    If Len(t) > 0 Then t = t & vbCrLf & vbCrLf
    t = t & who & ": " & s
    If Len(t) > HISTORY_BOX_MAX Then
        ' an MSForms text box cannot hold more than ~32767 characters: drop the oldest part
        t = Right$(t, HISTORY_BOX_MAX)
        p = InStr(t, vbCrLf & vbCrLf)
        If p > 0 Then t = Mid$(t, p + 4)
        t = RU("[... \u043D\u0430\u0447\u0430\u043B\u043E \u043F\u0435\u0440\u0435\u043F\u0438\u0441\u043A\u0438 \u0443\u0431\u0440\u0430\u043D\u043E \u0438\u0437 \u043E\u043A\u043D\u0430; \u0447\u0442\u043E\u0431\u044B \u0441\u043E\u0445\u0440\u0430\u043D\u0438\u0442\u044C \u0432\u0435\u0441\u044C \u0442\u0435\u043A\u0441\u0442, \u043D\u0430\u0436\u0438\u043C\u0430\u0439\u0442\u0435 \u00AB\u0412 \u043B\u0438\u0441\u0442\u00BB \u0437\u0430\u0440\u0430\u043D\u0435\u0435]") & vbCrLf & vbCrLf & t
    End If
    txtHistory.Text = t
    On Error Resume Next
    txtHistory.SetFocus
    txtHistory.SelStart = Len(txtHistory.Text)
    txtInput.SetFocus
End Sub
